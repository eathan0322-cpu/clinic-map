    // 監聽 Firebase 雲端資料變更（零延遲即時同步）
    clinicsRef.on('value', (snapshot) => {
      const data = snapshot.val();
      if (!data) {
        // 如果雲端為空，自動載入 40 家初始資料並初始化
        initialVerifiedClinics.forEach(c => {
          clinicsRef.child(c.id).set(c);
        });
        return;
      }
      clinics = Object.values(data);
      renderMap();
      document.getElementById('syncStatus').innerText = '🟢 雲端同步完成';
    }, (error) => {
      document.getElementById('syncStatus').innerText = '🔴 連線異常';
      console.error(error);
    });

    function detectDistrict(addr) {
      const districts = ['板橋區', '新莊區', '中和區', '永和區', '三重區', '蘆洲區', '土城區', '新店區', '汐止區', '樹林區', '三峽區', '鶯歌區', '林口區', '五股區', '泰山區', '淡水區'];
      for (let d of districts) {
        if (addr.includes(d) || addr.includes(d.replace('區', ''))) {
          return d;
        }
      }
      return '其他';
    }

    function renderMap() {
      markers.forEach(m => map.removeLayer(m));
      markers = [];

      const visibleClinics = currentFilter === 'ALL' 
        ? clinics 
        : clinics.filter(c => c.district === currentFilter);

      visibleClinics.forEach(item => {
        const marker = L.circleMarker([item.lat, item.lng], {
          radius: 12,
          fillColor: colorMap[item.status] || '#ef4444',
          color: '#ffffff',
          weight: 2,
          opacity: 1,
          fillOpacity: 0.9
        }).addTo(map);

        const gmapNavUrl = `https://www.google.com/maps/dir/?api=1&destination=${encodeURIComponent(item.name + ' ' + item.address)}&travelmode=driving`;
        const gmapSearchUrl = `https://www.google.com/search?q=${encodeURIComponent(item.name + ' ' + (item.district || '板橋'))}`;

        const popupContent = `
          <div style="font-size:14px; min-width: 220px;">
            <div style="font-size:11px; color:#38bdf8; font-weight:bold;">${item.district || '新北市'}</div>
            <strong style="font-size:16px;">${item.name}</strong><br>
            <span style="font-size:12px; color:#94a3b8;">${item.address || ''}</span><br>
            <b style="color:#38bdf8;">設備：</b>${item.equipment || '未填'}<br>
            <b style="color:#38bdf8;">情報：</b>${item.notes || '無'}<br>
            
            <div class="popup-btn-row">
              <button class="popup-btn" onclick="startEditClinic('${item.id}')" style="background:#2563eb; color:white;">✏️ 編輯情報</button>
              <a class="popup-btn" href="${gmapSearchUrl}" target="_blank" style="background:#475569; color:white;">🔍 Google 查核</a>
              <a class="popup-btn" href="${gmapNavUrl}" target="_blank" style="background:#0284c7; color:white;">🚗 店家導航</a>
              <button class="popup-btn" onclick="deleteClinic('${item.id}')" style="background:#ef4444; color:white;">🗑️ 刪除標記</button>
            </div>
          </div>
        `;
        marker.bindPopup(popupContent);
        markers.push(marker);
      });

      updateStats(visibleClinics);
    }

    function updateStats(filteredList) {
      const scopeName = currentFilter === 'ALL' ? '新北市全區' : currentFilter;
      document.getElementById('currentScopeTitle').innerText = `📍 ${scopeName}解鎖度`;
      document.getElementById('totalCount').innerHTML = `${filteredList.length} <span>家目標</span>`;
      
      const bar = document.getElementById('progressBar');
      bar.innerHTML = '';
      if (filteredList.length === 0) return;

      const counts = { red: 0, yellow: 0, orange: 0, green: 0, purple: 0 };
      filteredList.forEach(c => counts[c.status] = (counts[c.status] || 0) + 1);

      for (let s in counts) {
        if (counts[s] > 0) {
          const seg = document.createElement('div');
          seg.className = `prog-${s}`;
          seg.style.width = `${(counts[s] / filteredList.length) * 100}%`;
          bar.appendChild(seg);
        }
      }
    }

    window.startEditClinic = function(id) {
      const target = clinics.find(c => c.id === id);
      if (!target) return;

      document.getElementById('editingId').value = target.id;
      document.getElementById('name').value = target.name;
      document.getElementById('address').value = target.address;
      document.getElementById('status').value = target.status;
      document.getElementById('equipment').value = target.equipment || '';
      document.getElementById('notes').value = target.notes || '';

      document.getElementById('formTitle').innerText = `✏️ 正在編輯：${target.name}`;
      document.getElementById('cancelEditBtn').style.display = 'block';

      const submitBtn = document.getElementById('submitBtn');
      submitBtn.innerText = '💾 儲存並同步至雲端';
      submitBtn.classList.add('btn-update');

      document.getElementById('sidebar').scrollTop = 120;
    };

    window.resetFormToCreate = function() {
      document.getElementById('editingId').value = '';
      document.getElementById('clinicForm').reset();
      document.getElementById('formTitle').innerText = '⚔️ 登記探勘目標';
      document.getElementById('cancelEditBtn').style.display = 'none';

      const submitBtn = document.getElementById('submitBtn');
      submitBtn.innerText = '定位並存入雲端';
      submitBtn.classList.remove('btn-update');

    window.filterDistrict = function() {
      currentFilter = document.getElementById('districtFilter').value;
      renderMap();

      if (districtCoords[currentFilter]) {
        map.flyTo(districtCoords[currentFilter].center, districtCoords[currentFilter].zoom, {
          duration: 1.2
        });
      }
    };

    document.getElementById('clinicForm').addEventListener('submit', async (e) => {
      e.preventDefault();
      const editingId = document.getElementById('editingId').value;
      const submitBtn = document.getElementById('submitBtn');
      const address = document.getElementById('address').value;
      const name = document.getElementById('name').value;
      const detectedDistrict = detectDistrict(address);

      // 編輯更新
      if (editingId) {
        const target = clinics.find(c => c.id === editingId);
        if (target) {
          const updatedClinic = {
            ...target,
            name: name,
            address: address,
            district: detectedDistrict,
            status: document.getElementById('status').value,
            equipment: document.getElementById('equipment').value,
            notes: document.getElementById('notes').value
          };
          
          await clinicsRef.child(editingId).set(updatedClinic);
          map.closePopup();
          resetFormToCreate();
        }
        return;
      }

      // 新增目標（聯網查座標）
      submitBtn.disabled = true;
      submitBtn.innerText = '正在定位並儲存雲端...';

      try {
        const res = await fetch(`https://nominatim.openstreetmap.org/search?format=json&q=${encodeURIComponent(address)}`);
        const data = await res.json();

        if (data && data.length > 0) {
          const newId = 'c_' + Date.now();
          const lat = parseFloat(data[0].lat);
          const lng = parseFloat(data[0].lon);

          const newClinic = {
            id: newId,
            district: detectedDistrict,
            name: name,
            address: address,
            lat: lat,
            lng: lng,
            status: document.getElementById('status').value,
            equipment: document.getElementById('equipment').value,
            notes: document.getElementById('notes').value
          };

          await clinicsRef.child(newId).set(newClinic);
          
          if (currentFilter !== 'ALL' && currentFilter !== detectedDistrict) {
            document.getElementById('districtFilter').value = detectedDistrict;
            currentFilter = detectedDistrict;
          }

          map.setView([lat, lng], 16);
          e.target.reset();
        } else {
          alert('找不到這個地址，請確認有填寫行政區！');
        }
      } catch (err) {
        alert('連線失敗，請檢查網路連線。');
      } finally {
        submitBtn.disabled = false;
        submitBtn.innerText = '定位並存入雲端';
      }
    });

    window.deleteClinic = function(id) {

        clinicsRef.child(id).remove();
      }
    };

    document.getElementById('districtFilter').value = '板橋區';
